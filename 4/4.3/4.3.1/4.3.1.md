# 4.3.1. Scripts de creación y carga de datos de la base de datos relacional

Los scripts siguientes crean la base de datos `RutaAcademica` en **Microsoft SQL Server 2022** y cargan un conjunto de datos de prueba que cubre todos los módulos del sistema. Se presentan en dos partes: DDL (definición de estructura) y DML (inserción de datos).

> **Convención de ejecución:** los scripts deben ejecutarse en orden. Cada bloque está precedido por un comentario que identifica la tabla y el módulo al que pertenece.

---

## Parte 1 — DDL: Creación de la base de datos y las tablas

```sql
-- ============================================================
-- RutaAcademica — DDL completo
-- SGBD: Microsoft SQL Server 2022 Developer Edition
-- Collation: Modern_Spanish_CI_AI
-- Responsable DDL: Equipo AcadémicaTech (Grupo 3)
-- ============================================================

-- Crear la base de datos
IF NOT EXISTS (SELECT name FROM sys.databases WHERE name = N'RutaAcademica')
BEGIN
    CREATE DATABASE RutaAcademica
        COLLATE Modern_Spanish_CI_AI;
END;
GO

USE RutaAcademica;
GO

-- ============================================================
-- MÓDULO TRANSVERSAL — Seguridad, configuración y auditoría
-- ============================================================

-- Tabla: usuario
CREATE TABLE usuario (
    usuario_id          INT             IDENTITY(1,1)   NOT NULL,
    correo_institucional VARCHAR(100)                   NOT NULL,
    contrasena_hash     VARCHAR(255)                    NOT NULL,
    rol                 VARCHAR(20)                     NOT NULL,
    activo              BIT                             NOT NULL    DEFAULT 1,
    CONSTRAINT PK_usuario                   PRIMARY KEY (usuario_id),
    CONSTRAINT UQ_usuario_correo            UNIQUE      (correo_institucional),
    CONSTRAINT CK_usuario_rol               CHECK       (rol IN ('ESTUDIANTE','DOCENTE','GESTION','ADMIN'))
);
GO

-- ============================================================
-- MÓDULO 1 — Estructura académica
-- ============================================================

-- Tabla: facultad
CREATE TABLE facultad (
    facultad_id INT             IDENTITY(1,1)   NOT NULL,
    codigo      VARCHAR(20)                     NOT NULL,
    nombre      VARCHAR(100)                    NOT NULL,
    CONSTRAINT PK_facultad          PRIMARY KEY (facultad_id),
    CONSTRAINT UQ_facultad_codigo   UNIQUE      (codigo)
);
GO

-- Tabla: plan_estudio
CREATE TABLE plan_estudio (
    plan_estudio_id INT             IDENTITY(1,1)   NOT NULL,
    facultad_id     INT                             NOT NULL,
    nombre_carrera  VARCHAR(100)                    NOT NULL,
    total_creditos  INT                             NOT NULL,
    numero_ciclos   INT                             NOT NULL,
    estado          VARCHAR(20)                     NOT NULL,
    CONSTRAINT PK_plan_estudio          PRIMARY KEY (plan_estudio_id),
    CONSTRAINT FK_plan_estudio_facultad FOREIGN KEY (facultad_id) REFERENCES facultad(facultad_id),
    CONSTRAINT CK_plan_estudio_estado   CHECK       (estado IN ('ACTIVO','INACTIVO'))
);
GO

-- Tabla: curso
CREATE TABLE curso (
    curso_id            INT         IDENTITY(1,1)   NOT NULL,
    codigo              VARCHAR(20)                 NOT NULL,
    nombre              VARCHAR(100)                NOT NULL,
    creditos            INT                         NOT NULL,
    horas_teoria        INT                         NOT NULL,
    horas_practica      INT                         NOT NULL,
    horas_laboratorio   INT                         NOT NULL,
    estado              VARCHAR(20)                 NOT NULL,
    CONSTRAINT PK_curso             PRIMARY KEY (curso_id),
    CONSTRAINT UQ_curso_codigo      UNIQUE      (codigo),
    CONSTRAINT CK_curso_estado      CHECK       (estado IN ('ACTIVO','INACTIVO'))
);
GO

-- Tabla: plan_curso (relación N:M entre plan_estudio y curso)
CREATE TABLE plan_curso (
    plan_estudio_id INT     NOT NULL,
    curso_id        INT     NOT NULL,
    ciclo_sugerido  INT     NOT NULL,
    es_obligatorio  BIT     NOT NULL,
    CONSTRAINT PK_plan_curso                    PRIMARY KEY (plan_estudio_id, curso_id),
    CONSTRAINT FK_plan_curso_plan_estudio        FOREIGN KEY (plan_estudio_id) REFERENCES plan_estudio(plan_estudio_id),
    CONSTRAINT FK_plan_curso_curso              FOREIGN KEY (curso_id)         REFERENCES curso(curso_id)
);
GO

-- Tabla: prerrequisito (relación reflexiva en curso)
CREATE TABLE prerrequisito (
    curso_id            INT NOT NULL,
    curso_requisito_id  INT NOT NULL,
    CONSTRAINT PK_prerrequisito                     PRIMARY KEY (curso_id, curso_requisito_id),
    CONSTRAINT FK_prerrequisito_curso               FOREIGN KEY (curso_id)           REFERENCES curso(curso_id),
    CONSTRAINT FK_prerrequisito_curso_req           FOREIGN KEY (curso_requisito_id) REFERENCES curso(curso_id),
    CONSTRAINT CK_prerrequisito_no_recursivo        CHECK       (curso_id <> curso_requisito_id)
);
GO

-- Tabla: ciclo_academico
CREATE TABLE ciclo_academico (
    ciclo_academico_id          INT         IDENTITY(1,1)   NOT NULL,
    periodo                     VARCHAR(10)                 NOT NULL,
    fecha_inicio                DATE                        NOT NULL,
    fecha_fin                   DATE                        NOT NULL,
    fecha_inicio_matricula      DATETIME                    NOT NULL,
    fecha_fin_matricula         DATETIME                    NOT NULL,
    fecha_inicio_evaluaciones   DATE                        NOT NULL,
    fecha_fin_evaluaciones      DATE                        NOT NULL,
    estado                      VARCHAR(20)                 NOT NULL,
    CONSTRAINT PK_ciclo_academico           PRIMARY KEY (ciclo_academico_id),
    CONSTRAINT UQ_ciclo_academico_periodo   UNIQUE      (periodo),
    CONSTRAINT CK_ciclo_academico_estado    CHECK       (estado IN ('PLANIFICADO','ACTIVO','CERRADO'))
);
GO

-- Tabla: turno_matricula
CREATE TABLE turno_matricula (
    turno_matricula_id  INT             IDENTITY(1,1)   NOT NULL,
    ciclo_academico_id  INT                             NOT NULL,
    numero_turno        INT                             NOT NULL,
    promedio_minimo     DECIMAL(4,2)                    NOT NULL,
    promedio_maximo     DECIMAL(4,2)                    NOT NULL,
    tope_creditos       INT                             NOT NULL,
    fecha_hora_inicio   DATETIME                        NOT NULL,
    fecha_hora_fin      DATETIME                        NOT NULL,
    CONSTRAINT PK_turno_matricula                   PRIMARY KEY (turno_matricula_id),
    CONSTRAINT FK_turno_matricula_ciclo             FOREIGN KEY (ciclo_academico_id) REFERENCES ciclo_academico(ciclo_academico_id),
    CONSTRAINT UQ_turno_matricula_ciclo_numero      UNIQUE      (ciclo_academico_id, numero_turno)
);
GO

-- Tabla: docente
CREATE TABLE docente (
    docente_id          INT         IDENTITY(1,1)   NOT NULL,
    usuario_id          INT                         NOT NULL,
    facultad_id         INT                         NOT NULL,
    codigo              VARCHAR(20)                 NOT NULL,
    nombres             VARCHAR(100)                NOT NULL,
    apellidos           VARCHAR(100)                NOT NULL,
    regimen_dedicacion  VARCHAR(30)                 NOT NULL,
    categoria           VARCHAR(30)                 NOT NULL,
    CONSTRAINT PK_docente                   PRIMARY KEY (docente_id),
    CONSTRAINT FK_docente_usuario           FOREIGN KEY (usuario_id)   REFERENCES usuario(usuario_id),
    CONSTRAINT FK_docente_facultad          FOREIGN KEY (facultad_id)  REFERENCES facultad(facultad_id),
    CONSTRAINT UQ_docente_usuario_id        UNIQUE      (usuario_id),
    CONSTRAINT UQ_docente_codigo            UNIQUE      (codigo),
    CONSTRAINT CK_docente_regimen           CHECK       (regimen_dedicacion IN ('TIEMPO_COMPLETO','TIEMPO_PARCIAL')),
    CONSTRAINT CK_docente_categoria         CHECK       (categoria IN ('AUXILIAR','ASOCIADO','PRINCIPAL'))
);
GO

-- Tabla: aula
CREATE TABLE aula (
    aula_id     INT         IDENTITY(1,1)   NOT NULL,
    facultad_id INT                         NOT NULL,
    codigo      VARCHAR(20)                 NOT NULL,
    nombre      VARCHAR(100)                NOT NULL,
    ubicacion   VARCHAR(150)                NOT NULL,
    capacidad   INT                         NOT NULL,
    tipo        VARCHAR(20)                 NOT NULL,
    CONSTRAINT PK_aula              PRIMARY KEY (aula_id),
    CONSTRAINT FK_aula_facultad     FOREIGN KEY (facultad_id) REFERENCES facultad(facultad_id),
    CONSTRAINT UQ_aula_codigo       UNIQUE      (codigo),
    CONSTRAINT CK_aula_tipo         CHECK       (tipo IN ('TEORIA','LABORATORIO','TALLER'))
);
GO

-- Tabla: seccion
CREATE TABLE seccion (
    seccion_id          INT         IDENTITY(1,1)   NOT NULL,
    curso_id            INT                         NOT NULL,
    ciclo_academico_id  INT                         NOT NULL,
    docente_id          INT                         NOT NULL,
    aula_id             INT                         NOT NULL,
    codigo_seccion      VARCHAR(10)                 NOT NULL,
    vacantes_totales    INT                         NOT NULL,
    vacantes_disponibles INT                        NOT NULL,
    modalidad           VARCHAR(20)                 NOT NULL,
    CONSTRAINT PK_seccion                       PRIMARY KEY (seccion_id),
    CONSTRAINT FK_seccion_curso                 FOREIGN KEY (curso_id)           REFERENCES curso(curso_id),
    CONSTRAINT FK_seccion_ciclo_academico       FOREIGN KEY (ciclo_academico_id) REFERENCES ciclo_academico(ciclo_academico_id),
    CONSTRAINT FK_seccion_docente               FOREIGN KEY (docente_id)         REFERENCES docente(docente_id),
    CONSTRAINT FK_seccion_aula                  FOREIGN KEY (aula_id)            REFERENCES aula(aula_id),
    CONSTRAINT UQ_seccion_curso_ciclo_codigo    UNIQUE      (curso_id, ciclo_academico_id, codigo_seccion),
    CONSTRAINT CK_seccion_vacantes_disp         CHECK       (vacantes_disponibles >= 0 AND vacantes_disponibles <= vacantes_totales),
    CONSTRAINT CK_seccion_modalidad             CHECK       (modalidad IN ('PRESENCIAL','VIRTUAL','HIBRIDA'))
);
GO

-- Crear índice para consultas de vacantes en tiempo real (RNF-01)
CREATE NONCLUSTERED INDEX IX_seccion_vacantes_disponibles
    ON seccion(vacantes_disponibles)
    INCLUDE (vacantes_totales, modalidad);
GO

-- Tabla: horario_seccion
CREATE TABLE horario_seccion (
    horario_seccion_id  INT         IDENTITY(1,1)   NOT NULL,
    seccion_id          INT                         NOT NULL,
    dia_semana          TINYINT                     NOT NULL,
    hora_inicio         TIME                        NOT NULL,
    hora_fin            TIME                        NOT NULL,
    tipo_sesion         VARCHAR(20)                 NOT NULL,
    CONSTRAINT PK_horario_seccion           PRIMARY KEY (horario_seccion_id),
    CONSTRAINT FK_horario_seccion_seccion   FOREIGN KEY (seccion_id) REFERENCES seccion(seccion_id),
    CONSTRAINT CK_horario_seccion_dia       CHECK       (dia_semana BETWEEN 1 AND 6),
    CONSTRAINT CK_horario_seccion_horas     CHECK       (hora_fin > hora_inicio),
    CONSTRAINT CK_horario_seccion_tipo      CHECK       (tipo_sesion IN ('TEORIA','PRACTICA','LABORATORIO'))
);
GO

-- Tabla: ampliacion_vacantes
CREATE TABLE ampliacion_vacantes (
    ampliacion_vacantes_id  INT             IDENTITY(1,1)   NOT NULL,
    seccion_id              INT                             NOT NULL,
    usuario_solicita_id     INT                             NOT NULL,
    usuario_aprueba_id      INT                             NULL,
    cantidad_solicitada     INT                             NOT NULL,
    cantidad_aprobada       INT                             NULL,
    motivo                  VARCHAR(255)                    NOT NULL,
    estado                  VARCHAR(20)                     NOT NULL,
    fecha_solicitud         DATETIME                        NOT NULL,

    CONSTRAINT PK_ampliacion_vacantes               PRIMARY KEY (ampliacion_vacantes_id),

    CONSTRAINT FK_ampliacion_vacantes_solicita       FOREIGN KEY (usuario_solicita_id) REFERENCES usuario(usuario_id),
    CONSTRAINT FK_ampliacion_vacantes_aprueba        FOREIGN KEY (usuario_aprueba_id)  REFERENCES usuario(usuario_id),
    CONSTRAINT CK_ampliacion_vacantes_estado         CHECK       (estado IN ('PENDIENTE','APROBADA','RECHAZADA'))

GO

    motivo                  VARCHAR(30)                     NOT NULL,
    CONSTRAINT PK_demanda_insatisfecha              PRIMARY KEY (demanda_insatisfecha_id),

-- Tabla: reserva_vacante
CREATE TABLE reserva_vacante (
    reserva_vacante_id  INT         IDENTITY(1,1)   NOT NULL,
    estudiante_id       INT                         NOT NULL,
    seccion_id          INT                         NOT NULL,
    fecha_reserva       DATETIME                    NOT NULL,
    fecha_vencimiento   DATETIME                    NOT NULL,
    estado              VARCHAR(20)                 NOT NULL,
    CONSTRAINT PK_reserva_vacante               PRIMARY KEY (reserva_vacante_id),
    CONSTRAINT FK_reserva_vacante_estudiante    FOREIGN KEY (estudiante_id) REFERENCES estudiante(estudiante_id),
    CONSTRAINT FK_reserva_vacante_seccion       FOREIGN KEY (seccion_id)    REFERENCES seccion(seccion_id),
    CONSTRAINT CK_reserva_vacante_vencimiento   CHECK       (fecha_vencimiento > fecha_reserva),
    CONSTRAINT CK_reserva_vacante_estado        CHECK       (estado IN ('ACTIVA','CONFIRMADA','VENCIDA','CANCELADA'))
);
GO

-- Tabla: matricula
CREATE TABLE matricula (
    matricula_id            INT             IDENTITY(1,1)   NOT NULL,
    estudiante_id           INT                             NOT NULL,
    seccion_id              INT                             NOT NULL,
    reserva_vacante_id      INT                             NULL,
    usuario_registra_id     INT                             NOT NULL,
    tipo_registro           VARCHAR(20)                     NOT NULL,
    motivo_rectificacion    VARCHAR(255)                    NULL,
    numero_matricula_curso  INT                             NOT NULL,
    estado                  VARCHAR(20)                     NOT NULL,
    nota_final              DECIMAL(4,2)                    NULL,
    fecha_matricula         DATETIME                        NOT NULL,
    CONSTRAINT PK_matricula                         PRIMARY KEY (matricula_id),
    CONSTRAINT FK_matricula_estudiante              FOREIGN KEY (estudiante_id)       REFERENCES estudiante(estudiante_id),
    CONSTRAINT FK_matricula_seccion                 FOREIGN KEY (seccion_id)          REFERENCES seccion(seccion_id),
    CONSTRAINT FK_matricula_reserva_vacante         FOREIGN KEY (reserva_vacante_id)  REFERENCES reserva_vacante(reserva_vacante_id),
    CONSTRAINT FK_matricula_usuario_registra        FOREIGN KEY (usuario_registra_id) REFERENCES usuario(usuario_id),
    CONSTRAINT UQ_matricula_reserva_vacante_id      UNIQUE      (reserva_vacante_id),
    CONSTRAINT UQ_matricula_estudiante_seccion      UNIQUE      (estudiante_id, seccion_id),
    CONSTRAINT CK_matricula_tipo_registro           CHECK       (tipo_registro IN ('REGULAR','RECTIFICACION')),
    CONSTRAINT CK_matricula_estado                  CHECK       (estado IN ('MATRICULADO','RETIRADO','APROBADO','DESAPROBADO')),
    CONSTRAINT CK_matricula_nota_final              CHECK       (nota_final IS NULL OR nota_final BETWEEN 0 AND 20),
    CONSTRAINT CK_matricula_numero_curso            CHECK       (numero_matricula_curso BETWEEN 1 AND 4)
);
GO

-- MÓDULO TRANSVERSAL — Configuración y auditoría
-- ============================================================

-- Tabla: parametro_sistema
CREATE TABLE parametro_sistema (
    parametro_sistema_id    INT             IDENTITY(1,1)   NOT NULL,
    clave                   VARCHAR(100)                    NOT NULL,
    valor                   VARCHAR(255)                    NOT NULL,
    descripcion             VARCHAR(500)                    NOT NULL,
    usuario_modifica_id     INT                             NOT NULL,
    fecha_modificacion      DATETIME                        NOT NULL,
    CONSTRAINT PK_parametro_sistema                 PRIMARY KEY (parametro_sistema_id),
    CONSTRAINT FK_parametro_sistema_usuario         FOREIGN KEY (usuario_modifica_id) REFERENCES usuario(usuario_id),
    CONSTRAINT UQ_parametro_sistema_clave           UNIQUE      (clave)
);
GO

-- Tabla: auditoria
CREATE TABLE auditoria (
    auditoria_id        INT             IDENTITY(1,1)   NOT NULL,
    usuario_id          INT                             NOT NULL,
    tabla_afectada      VARCHAR(100)                    NOT NULL,
    registro_afectado   VARCHAR(100)                    NOT NULL,
    operacion           VARCHAR(20)                     NOT NULL,
    fecha_hora          DATETIME                        NOT NULL,
    detalle             VARCHAR(MAX)                    NULL,
    CONSTRAINT PK_auditoria                 PRIMARY KEY (auditoria_id),
    CONSTRAINT FK_auditoria_usuario         FOREIGN KEY (usuario_id) REFERENCES usuario(usuario_id),
    CONSTRAINT CK_auditoria_operacion       CHECK       (operacion IN ('INSERT','UPDATE','DELETE'))
);
GO
```
