# 4.3.2. Scripts de creación y carga de datos de la base de datos no relacional

Los scripts siguientes crean las colecciones `sesiones_tutoria` y `seguimiento_riesgo` en la base de datos `ruta_academica` de **MongoDB 7.0**, aplican la validación de esquema definida en la sección 4.2.2, crean los índices y cargan un conjunto de documentos de prueba coherentes con los datos de la sección 4.3.1.

> **Convención de ejecución:** los scripts se ejecutan en MongoDB Shell (`mongosh`) conectado a `localhost:27017`. Deben ejecutarse en orden: primero la creación de colecciones e índices, luego la carga de datos.

---

## Parte 1 — Creación de colecciones con validación de esquema e índices

```javascript
// ============================================================
// RutaAcademica — MongoDB: Creación de colecciones
// SGBD: MongoDB Community Server 7.0
// Base de datos: ruta_academica
// ============================================================

use("ruta_academica");

// ------------------------------------------------------------
// Colección: sesiones_tutoria
// Patrón: Embedded Document
// ------------------------------------------------------------
db.createCollection("sesiones_tutoria", {
  collation: { locale: "es", strength: 1 },
  validator: {
    $jsonSchema: {
      bsonType: "object",
      title: "Documento de sesiones de tutoría",
      required: [
        "asignacion_tutor_id", "ciclo_academico", "tutor",
        "estudiante", "fecha_asignacion", "estado_asignacion",
        "sesiones", "total_sesiones"
      ],
      properties: {
        asignacion_tutor_id: {
          bsonType: "int",
          minimum: 1,
          description: "ID de la asignación de tutor en SQL Server."
        },
        ciclo_academico: {
          bsonType: "object",
          required: ["ciclo_academico_id", "periodo", "fecha_inicio", "fecha_fin"],
          properties: {
            ciclo_academico_id: { bsonType: "int" },
            periodo: { bsonType: "string", pattern: "^[0-9]{4}-[12]$" },
            fecha_inicio: { bsonType: "date" },
            fecha_fin:    { bsonType: "date" }
          }
        },
        tutor: {
          bsonType: "object",
          required: ["docente_id", "codigo", "nombres", "apellidos"],
          properties: {
            docente_id:              { bsonType: "int" },
            codigo:                  { bsonType: "string" },
            nombres:                 { bsonType: "string" },
            apellidos:               { bsonType: "string" },
            correo_institucional:    { bsonType: "string" }
          }
        },
        estudiante: {
          bsonType: "object",
          required: ["estudiante_id", "codigo", "nombres", "apellidos", "ciclo_actual", "estado_academico"],
          properties: {
            estudiante_id:           { bsonType: "int" },
            codigo:                  { bsonType: "string" },
            nombres:                 { bsonType: "string" },
            apellidos:               { bsonType: "string" },
            correo_institucional:    { bsonType: "string" },
            plan_estudio:            { bsonType: "string" },
            ciclo_actual:            { bsonType: "int", minimum: 1 },
            estado_academico: {
              bsonType: "string",
              enum: ["REGULAR", "REPITENCIA_2", "REPITENCIA_3", "SEPARADO"]
            }
          }
        },
        fecha_asignacion:    { bsonType: "date" },
        estado_asignacion: {
          bsonType: "string",
          enum: ["VIGENTE", "FINALIZADA"]
        },
        sesiones: {
          bsonType: "array",
          items: {
            bsonType: "object",
            required: ["fecha", "duracion_minutos", "temas_tratados"],
            properties: {
              sesion_tutoria_id:   { bsonType: "int" },
              fecha:               { bsonType: "date" },
              duracion_minutos:    { bsonType: "int", minimum: 15 },
              temas_tratados:      { bsonType: "string", minLength: 1 },
              compromisos:         { bsonType: ["string", "null"] }
            }
          }
        },
        total_sesiones:       { bsonType: "int", minimum: 0 },
        fecha_ultima_sesion:  { bsonType: ["date", "null"] }
      }
    }
  },
  validationLevel: "strict",
  validationAction: "error"
});

// Índices de sesiones_tutoria
db.sesiones_tutoria.createIndex(
  { "asignacion_tutor_id": 1 },
  { unique: true, name: "uq_asignacion_tutor_id" }
);
db.sesiones_tutoria.createIndex(
  { "estudiante.estudiante_id": 1 },
  { name: "idx_est_id" }
);
db.sesiones_tutoria.createIndex(
  { "tutor.docente_id": 1 },
  { name: "idx_tutor_id" }
);
db.sesiones_tutoria.createIndex(
  { "ciclo_academico.periodo": 1 },
  { name: "idx_periodo" }
);
db.sesiones_tutoria.createIndex(
  { "estado_asignacion": 1 },
  { name: "idx_estado_asignacion" }
);

// ------------------------------------------------------------
// Colección: seguimiento_riesgo
// Patrón: Subset
// ------------------------------------------------------------
db.createCollection("seguimiento_riesgo", {
  collation: { locale: "es", strength: 1 },
  validator: {
    $jsonSchema: {
      bsonType: "object",
      title: "Documento de seguimiento de riesgo académico",
      required: [
        "matricula_id", "estudiante", "seccion", "ciclo",
        "indicador", "alertas", "historial_asistencia_reciente",
        "numero_matricula_curso", "fecha_actualizacion"
      ],
      properties: {
        matricula_id: { bsonType: "int", minimum: 1 },
        estudiante: {
          bsonType: "object",
          required: ["estudiante_id", "codigo", "nombres", "apellidos"],
          properties: {
            estudiante_id: { bsonType: "int" },
            codigo:        { bsonType: "string" },
            nombres:       { bsonType: "string" },
            apellidos:     { bsonType: "string" }
          }
        },
        seccion: {
          bsonType: "object",
          required: ["seccion_id", "curso_codigo", "curso_nombre", "codigo_seccion"],
          properties: {
            seccion_id:        { bsonType: "int" },
            curso_codigo:      { bsonType: "string" },
            curso_nombre:      { bsonType: "string" },
            codigo_seccion:    { bsonType: "string" },
            docente_nombres:   { bsonType: "string" },
            docente_apellidos: { bsonType: "string" }
          }
        },
        ciclo: {
          bsonType: "object",
          required: ["ciclo_academico_id", "periodo"],
          properties: {
            ciclo_academico_id: { bsonType: "int" },
            periodo:            { bsonType: "string", pattern: "^[0-9]{4}-[12]$" }
          }
        },
        indicador: {
          bsonType: "object",
          required: ["nivel_riesgo", "fecha_calculo"],
          properties: {
            nota_promedio_parcial:  { bsonType: ["decimal", "null"] },
            porcentaje_asistencia:  { bsonType: ["decimal", "null"] },
            nivel_riesgo: {
              bsonType: "string",
              enum: ["BAJO", "MEDIO", "ALTO"]
            },
            fecha_calculo: { bsonType: "date" }
          }
        },
        alertas: {
          bsonType: "array",
          items: {
            bsonType: "object",
            required: ["motivo", "estado", "fecha_generacion"],
            properties: {
              alerta_riesgo_id:  { bsonType: "int" },
              motivo: {
                bsonType: "string",
                enum: ["NOTA_BAJA", "INASISTENCIA", "AMBOS"]
              },
              estado: {
                bsonType: "string",
                enum: ["ACTIVA", "ATENDIDA", "DESCARTADA"]
              },
              fecha_generacion: { bsonType: "date" }
            }
          }
        },
        historial_asistencia_reciente: {
          bsonType: "array",
          maxItems: 10,
          items: {
            bsonType: "object",
            required: ["fecha_sesion", "estado"],
            properties: {
              fecha_sesion: { bsonType: "date" },
              estado: {
                bsonType: "string",
                enum: ["PRESENTE", "AUSENTE", "TARDANZA", "JUSTIFICADO"]
              }
            }
          }
        },
        numero_matricula_curso: { bsonType: "int", minimum: 1, maximum: 4 },
        fecha_actualizacion:    { bsonType: "date" }
      }
    }
  },
  validationLevel: "strict",
  validationAction: "error"
});

// Índices de seguimiento_riesgo
db.seguimiento_riesgo.createIndex(
  { "matricula_id": 1 },
  { unique: true, name: "uq_matricula_id" }
);
db.seguimiento_riesgo.createIndex(
  { "estudiante.estudiante_id": 1 },
  { name: "idx_estudiante_id" }
);
db.seguimiento_riesgo.createIndex(
  { "indicador.nivel_riesgo": 1 },
  { name: "idx_nivel_riesgo" }
);
db.seguimiento_riesgo.createIndex(
  { "seccion.seccion_id": 1 },
  { name: "idx_seccion_id" }
);
db.seguimiento_riesgo.createIndex(
  { "ciclo.periodo": 1 },
  { name: "idx_periodo" }
);
```
