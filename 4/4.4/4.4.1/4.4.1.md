# 4.4.1. Consultas para la base de datos relacional

Las consultas siguientes se implementaron en **T-SQL** sobre la base de datos `RutaAcademica` en Microsoft SQL Server 2022. Cada integrante del equipo es responsable de cinco consultas alineadas con su módulo. Las consultas cubren los requisitos funcionales identificados en la sección 2.2.1.

## Módulo 1 — Estructura académica (Responsable: Eustaquio)

### C-01. Listar cursos de un plan de estudios con su ciclo sugerido y prerrequisitos

**Propósito:** Permite a gestión académica visualizar la malla completa de una carrera, incluyendo el ciclo recomendado y los prerrequisitos de cada curso (RF-02).

```sql
-- C-01: Malla curricular con prerrequisitos
SELECT
    pc.ciclo_sugerido,
    c.codigo                    AS curso_codigo,
    c.nombre                    AS curso_nombre,
    c.creditos,
    CASE pc.es_obligatorio
        WHEN 1 THEN 'Obligatorio'
        ELSE 'Electivo'
    END                         AS tipo,
    ISNULL(cr.codigo, '—')      AS prereq_codigo,
    ISNULL(cr.nombre, '—')      AS prereq_nombre
FROM plan_curso pc
    INNER JOIN plan_estudio pe  ON pe.plan_estudio_id = pc.plan_estudio_id
    INNER JOIN curso c          ON c.curso_id         = pc.curso_id
    LEFT  JOIN prerrequisito pr ON pr.curso_id        = c.curso_id
    LEFT  JOIN curso cr         ON cr.curso_id        = pr.curso_requisito_id
WHERE pe.nombre_carrera = 'Ingeniería de Software'
  AND pe.estado         = 'ACTIVO'
ORDER BY pc.ciclo_sugerido, c.nombre;
```

### C-02. Consultar secciones abiertas por curso y ciclo con disponibilidad de vacantes

**Propósito:** Permite al estudiante verificar las secciones disponibles antes de su turno (RF-11).

```sql
-- C-02: Disponibilidad de vacantes por curso y ciclo
SELECT
    c.codigo                                        AS curso_codigo,
    c.nombre                                        AS curso_nombre,
    s.codigo_seccion,
    d.nombres + ' ' + d.apellidos                   AS docente,
    a.codigo                                        AS aula,
    s.vacantes_disponibles,
    s.vacantes_totales,
    s.modalidad,
    STRING_AGG(
        CASE hs.dia_semana
            WHEN 1 THEN 'Lun' WHEN 2 THEN 'Mar' WHEN 3 THEN 'Mie'
            WHEN 4 THEN 'Jue' WHEN 5 THEN 'Vie' WHEN 6 THEN 'Sab'
        END + ' ' +
        FORMAT(CAST(hs.hora_inicio AS DATETIME), 'HH:mm') + '-' +
        FORMAT(CAST(hs.hora_fin    AS DATETIME), 'HH:mm'),
        ' | '
    ) WITHIN GROUP (ORDER BY hs.dia_semana, hs.hora_inicio) AS horario
FROM seccion s
    INNER JOIN curso c              ON c.curso_id            = s.curso_id
    INNER JOIN ciclo_academico ca   ON ca.ciclo_academico_id = s.ciclo_academico_id
    INNER JOIN docente d            ON d.docente_id          = s.docente_id
    INNER JOIN aula a               ON a.aula_id             = s.aula_id
    LEFT  JOIN horario_seccion hs   ON hs.seccion_id         = s.seccion_id
WHERE ca.periodo = '2026-2'
  AND c.codigo   = 'SW609'
GROUP BY
    c.codigo, c.nombre, s.codigo_seccion,
    d.nombres, d.apellidos, a.codigo,
    s.vacantes_disponibles, s.vacantes_totales, s.modalidad
ORDER BY s.vacantes_disponibles DESC;
```

### C-05. Validar si un estudiante cumple prerrequisitos para un curso

**Propósito:** Verifica de forma automática si el estudiante puede matricularse en un curso, revisando que haya aprobado todos sus prerrequisitos (RF-13, RN-03). Se presenta como procedimiento almacenado.

```sql
-- C-05: Procedimiento para validar prerrequisitos
CREATE OR ALTER PROCEDURE sp_validar_prerrequisitos
    @estudiante_id  INT,
    @curso_id       INT,
    @puede_matricular BIT OUTPUT,
    @mensaje          VARCHAR(500) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @prereqs_pendientes INT;

    -- Contar prerrequisitos no aprobados
    SELECT @prereqs_pendientes = COUNT(*)
    FROM prerrequisito pr
    WHERE pr.curso_id = @curso_id
      AND NOT EXISTS (
          SELECT 1
          FROM matricula m
              INNER JOIN seccion sec ON sec.seccion_id = m.seccion_id
          WHERE m.estudiante_id = @estudiante_id
            AND sec.curso_id    = pr.curso_requisito_id
            AND m.estado        = 'APROBADO'
      );

    IF @prereqs_pendientes = 0
    BEGIN
        SET @puede_matricular = 1;
        SET @mensaje = 'El estudiante cumple todos los prerrequisitos.';
    END
    ELSE
    BEGIN
        SET @puede_matricular = 0;
        SET @mensaje = CONCAT(
            'El estudiante tiene ', @prereqs_pendientes,
            ' prerrequisito(s) pendiente(s) para este curso.'
        );
    END;
END;
GO

-- Ejemplo de ejecución
DECLARE @puede BIT, @msg VARCHAR(500);
EXEC sp_validar_prerrequisitos
    @estudiante_id    = 1,
    @curso_id         = 1,
    @puede_matricular = @puede OUTPUT,
    @mensaje          = @msg   OUTPUT;
SELECT @puede AS puede_matricular, @msg AS mensaje;
GO
```

### C-08. Función: calcular créditos matriculados por estudiante en el ciclo activo

**Propósito:** Calcula el total de créditos que tiene un estudiante en el ciclo activo, para verificar que no supera el tope de su turno (RF-13, RN-02).

```sql
-- C-08: Función escalar de créditos matriculados
CREATE OR ALTER FUNCTION fn_creditos_matriculados (
    @estudiante_id      INT,
    @ciclo_academico_id INT
)
RETURNS INT
AS
BEGIN
    DECLARE @total_creditos INT;

    SELECT @total_creditos = ISNULL(SUM(c.creditos), 0)
    FROM matricula m
        INNER JOIN seccion s    ON s.seccion_id  = m.seccion_id
        INNER JOIN curso c      ON c.curso_id    = s.curso_id
    WHERE m.estudiante_id       = @estudiante_id
      AND s.ciclo_academico_id  = @ciclo_academico_id
      AND m.estado              = 'MATRICULADO';

    RETURN @total_creditos;
END;
GO

-- Ejemplo de uso
SELECT
    e.codigo,
    e.nombres + ' ' + e.apellidos      AS estudiante,
    dbo.fn_creditos_matriculados(e.estudiante_id, 2) AS creditos_matriculados
FROM estudiante e
ORDER BY creditos_matriculados DESC;
GO
```

### C-10. Procedimiento: registrar ampliación de vacantes y actualizar sección

**Propósito:** Registra una ampliación aprobada en la tabla `ampliacion_vacantes` y actualiza de forma atómica el contador de vacantes de la sección (RF-08, RN-08).

```sql
-- C-10: Procedimiento para aprobar ampliación de vacantes
CREATE OR ALTER PROCEDURE sp_aprobar_ampliacion
    @ampliacion_vacantes_id INT,
    @usuario_aprueba_id     INT,
    @cantidad_aprobada      INT
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRANSACTION;
    BEGIN TRY
        -- Actualizar la solicitud
        UPDATE ampliacion_vacantes
        SET
            cantidad_aprobada   = @cantidad_aprobada,
            usuario_aprueba_id  = @usuario_aprueba_id,
            estado              = 'APROBADA',
            fecha_resolucion    = GETDATE()
        WHERE ampliacion_vacantes_id = @ampliacion_vacantes_id
          AND estado                 = 'PENDIENTE';

        IF @@ROWCOUNT = 0
            THROW 50001, 'La solicitud no existe o ya fue resuelta.', 1;

        -- Actualizar vacantes en la sección
        UPDATE s
        SET vacantes_totales     = s.vacantes_totales     + @cantidad_aprobada,
            vacantes_disponibles = s.vacantes_disponibles + @cantidad_aprobada
        FROM seccion s
            INNER JOIN ampliacion_vacantes av
                ON av.seccion_id = s.seccion_id
        WHERE av.ampliacion_vacantes_id = @ampliacion_vacantes_id;

        -- Registrar en auditoría
        INSERT INTO auditoria (usuario_id, tabla_afectada, registro_afectado, operacion, fecha_hora, detalle)
        VALUES (@usuario_aprueba_id, 'ampliacion_vacantes',
                CAST(@ampliacion_vacantes_id AS VARCHAR),
                'UPDATE', GETDATE(),
                CONCAT('Ampliación aprobada. Vacantes añadidas: ', @cantidad_aprobada));

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        THROW;
    END CATCH;
END;
GO
```

### C-14. Procedimiento: confirmar matrícula con validación de reglas

**Propósito:** Confirma la matrícula del estudiante verificando prerrequisitos, tope de créditos y cruce de horario (RF-13, RN-01 a RN-04).

```sql
-- C-14: Procedimiento de confirmación de matrícula
CREATE OR ALTER PROCEDURE sp_confirmar_matricula
    @reserva_vacante_id     INT,
    @usuario_registra_id    INT,
    @resultado              BIT         OUTPUT,
    @mensaje                VARCHAR(500) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE
        @estudiante_id      INT,
        @seccion_id         INT,
        @ciclo_id           INT,
        @curso_id           INT,
        @cred_matriculados  INT,
        @tope_creditos      INT,
        @cred_nuevo         INT,
        @veces              INT;

    -- Leer datos de la reserva
    SELECT
        @estudiante_id = rv.estudiante_id,
        @seccion_id    = rv.seccion_id
    FROM reserva_vacante rv
    WHERE rv.reserva_vacante_id = @reserva_vacante_id
      AND rv.estado             = 'ACTIVA';

    IF @estudiante_id IS NULL
    BEGIN
        SET @resultado = 0;
        SET @mensaje   = 'La reserva no existe o no está activa.';
        RETURN;
    END;

    SELECT @ciclo_id = s.ciclo_academico_id,
           @curso_id = s.curso_id
    FROM seccion s WHERE s.seccion_id = @seccion_id;

    -- Validar prerrequisitos
    DECLARE @prereqs_pendientes INT = 0;
    EXEC sp_validar_prerrequisitos
        @estudiante_id    = @estudiante_id,
        @curso_id         = @curso_id,
        @puede_matricular = @resultado OUTPUT,
        @mensaje          = @mensaje   OUTPUT;

    IF @resultado = 0 RETURN;

    -- Validar tope de créditos
    SELECT @tope_creditos = tm.tope_creditos
    FROM turno_matricula tm
        INNER JOIN estudiante e ON e.promedio_ponderado BETWEEN tm.promedio_minimo AND tm.promedio_maximo
    WHERE tm.ciclo_academico_id = @ciclo_id
      AND e.estudiante_id       = @estudiante_id;

    SET @cred_matriculados = dbo.fn_creditos_matriculados(@estudiante_id, @ciclo_id);
    SELECT @cred_nuevo = c.creditos FROM seccion s INNER JOIN curso c ON c.curso_id = s.curso_id
    WHERE s.seccion_id = @seccion_id;

    IF @cred_matriculados + @cred_nuevo > ISNULL(@tope_creditos, 999)
    BEGIN
        SET @resultado = 0;
        SET @mensaje   = CONCAT('Supera el tope de créditos del turno (', @tope_creditos, ' cr.).');
        RETURN;
    END;

    -- Contar veces que lleva el curso
    SELECT @veces = COUNT(*) + 1
    FROM matricula m
        INNER JOIN seccion s ON s.seccion_id = m.seccion_id
    WHERE m.estudiante_id = @estudiante_id AND s.curso_id = @curso_id;

    -- Confirmar matrícula en transacción
    BEGIN TRANSACTION;
    BEGIN TRY
        INSERT INTO matricula
            (estudiante_id, seccion_id, reserva_vacante_id, usuario_registra_id,
             tipo_registro, numero_matricula_curso, estado, fecha_matricula)
        VALUES
            (@estudiante_id, @seccion_id, @reserva_vacante_id, @usuario_registra_id,
             'REGULAR', @veces, 'MATRICULADO', GETDATE());

        UPDATE reserva_vacante SET estado = 'CONFIRMADA'
        WHERE reserva_vacante_id = @reserva_vacante_id;

        COMMIT TRANSACTION;
        SET @resultado = 1;
        SET @mensaje   = 'Matrícula confirmada correctamente.';
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        SET @resultado = 0;
        SET @mensaje   = ERROR_MESSAGE();
    END CATCH;
END;
GO
```

### C-15. Función: obtener promedio ponderado acumulado de un estudiante

**Propósito:** Calcula el promedio ponderado acumulado de un estudiante a partir de sus matrículas con nota final aprobada, para determinar su turno en el siguiente ciclo (RF-04, RN-01).

```sql
-- C-15: Función de promedio ponderado acumulado
CREATE OR ALTER FUNCTION fn_promedio_ponderado (@estudiante_id INT)
RETURNS DECIMAL(4,2)
AS
BEGIN
    DECLARE @promedio DECIMAL(4,2);

    SELECT @promedio = CAST(
        ISNULL(
            SUM(c.creditos * m.nota_final) /
            NULLIF(SUM(c.creditos), 0),
            0
        ) AS DECIMAL(4,2)
    )
    FROM matricula m
        INNER JOIN seccion s ON s.seccion_id = m.seccion_id
        INNER JOIN curso c   ON c.curso_id   = s.curso_id
    WHERE m.estudiante_id = @estudiante_id
      AND m.nota_final IS NOT NULL
      AND m.estado IN ('APROBADO', 'DESAPROBADO');

    RETURN @promedio;
END;
GO

-- Uso
SELECT
    e.codigo,
    e.nombres + ' ' + e.apellidos    AS estudiante,
    e.promedio_ponderado             AS pp_almacenado,
    dbo.fn_promedio_ponderado(e.estudiante_id) AS pp_calculado
FROM estudiante e
ORDER BY pp_calculado DESC;
GO
```

## Módulo 4 — Evaluación, asistencia y riesgo (Responsable: Paucar)

### C-16. Panel de riesgo por sección: nota parcial, asistencia y nivel de riesgo

**Propósito:** Proporciona al docente/tutor el panel de seguimiento de todos los estudiantes de una sección (RF-22).

```sql
-- C-16: Panel de riesgo de una sección
SELECT
    e.codigo                        AS cod_estudiante,
    e.nombres + ' ' + e.apellidos   AS estudiante,
    e.estado_academico,
    ir.nota_promedio_parcial,
    ir.porcentaje_asistencia,
    ir.nivel_riesgo,
    FORMAT(ir.fecha_calculo, 'dd/MM/yyyy HH:mm') AS ultima_actualizacion,
    ISNULL(ar.motivo, '—')          AS alerta_motivo,
    ISNULL(ar.estado,  '—')         AS alerta_estado
FROM matricula m
    INNER JOIN estudiante e         ON e.estudiante_id       = m.estudiante_id
    LEFT  JOIN indicador_riesgo ir  ON ir.matricula_id       = m.matricula_id
    LEFT  JOIN alerta_riesgo ar     ON ar.matricula_id       = m.matricula_id
                                    AND ar.estado            = 'ACTIVA'
WHERE m.seccion_id = 1
ORDER BY ir.nivel_riesgo DESC, ir.nota_promedio_parcial ASC;
```

### C-17. Porcentaje de asistencia por estudiante y sección

**Propósito:** Calcula el porcentaje de asistencia de cada estudiante en una sección para detectar riesgo de inhabilitación (RF-18, RF-19, RN-11).

```sql
-- C-17: Porcentaje de asistencia calculado
SELECT
    e.codigo                        AS cod_estudiante,
    e.nombres + ' ' + e.apellidos   AS estudiante,
    COUNT(*)                        AS total_sesiones,
    SUM(CASE WHEN a.estado IN ('PRESENTE','TARDANZA','JUSTIFICADO') THEN 1 ELSE 0 END) AS presencias,
    SUM(CASE WHEN a.estado = 'AUSENTE' THEN 1 ELSE 0 END) AS ausencias,
    CAST(
        100.0 *
        SUM(CASE WHEN a.estado IN ('PRESENTE','TARDANZA','JUSTIFICADO') THEN 1 ELSE 0 END) /
        NULLIF(COUNT(*), 0)
    AS DECIMAL(5,2))                AS pct_asistencia,
    CASE
        WHEN CAST(
            100.0 *
            SUM(CASE WHEN a.estado IN ('PRESENTE','TARDANZA','JUSTIFICADO') THEN 1 ELSE 0 END) /
            NULLIF(COUNT(*), 0)
        AS DECIMAL(5,2)) < 70 THEN 'RIESGO'
        ELSE 'OK'
    END                             AS estado_asistencia
FROM asistencia a
    INNER JOIN matricula m          ON m.matricula_id  = a.matricula_id
    INNER JOIN estudiante e         ON e.estudiante_id = m.estudiante_id
    INNER JOIN sesion_clase sc      ON sc.sesion_clase_id = a.sesion_clase_id
WHERE m.seccion_id = 1
GROUP BY e.codigo, e.nombres, e.apellidos
ORDER BY pct_asistencia ASC;
```

### C-18. Notas promedio por evaluación de una sección

**Propósito:** Permite al docente comparar el rendimiento del grupo en cada instrumento de evaluación para ajustar su estrategia pedagógica (RF-17).

```sql
-- C-18: Estadísticas de notas por evaluación
SELECT
    ev.nombre               AS evaluacion,
    FORMAT(ev.fecha, 'dd/MM/yyyy') AS fecha,
    ev.peso_porcentual,
    COUNT(cal.calificacion_id)  AS estudiantes_evaluados,
    MIN(cal.nota)               AS nota_minima,
    MAX(cal.nota)               AS nota_maxima,
    CAST(AVG(cal.nota) AS DECIMAL(4,2)) AS nota_promedio,
    COUNT(CASE WHEN cal.nota < 10.5 THEN 1 END) AS bajo_umbral
FROM evaluacion ev
    LEFT JOIN calificacion cal ON cal.evaluacion_id = ev.evaluacion_id
WHERE ev.seccion_id = 1
GROUP BY ev.evaluacion_id, ev.nombre, ev.fecha, ev.peso_porcentual
ORDER BY ev.fecha;
```

### C-19. Alertas de riesgo activas por tutor

**Propósito:** Lista todas las alertas activas que debe atender un tutor, con el detalle del estudiante y el motivo (RF-20, RF-22).

```sql
-- C-19: Alertas activas de un tutor
SELECT
    d.nombres + ' ' + d.apellidos   AS tutor,
    e.codigo                        AS cod_estudiante,
    e.nombres + ' ' + e.apellidos   AS estudiante,
    c.codigo                        AS curso_codigo,
    s.codigo_seccion,
    ar.motivo,
    ir.nota_promedio_parcial,
    ir.porcentaje_asistencia,
    ir.nivel_riesgo,
    FORMAT(ar.fecha_generacion, 'dd/MM/yyyy HH:mm') AS fecha_alerta
FROM alerta_riesgo ar
    INNER JOIN matricula m          ON m.matricula_id       = ar.matricula_id
    INNER JOIN estudiante e         ON e.estudiante_id      = m.estudiante_id
    INNER JOIN seccion s            ON s.seccion_id         = m.seccion_id
    INNER JOIN curso c              ON c.curso_id           = s.curso_id
    LEFT  JOIN indicador_riesgo ir  ON ir.matricula_id      = m.matricula_id
    INNER JOIN docente d            ON d.docente_id         = ar.docente_id
WHERE ar.estado     = 'ACTIVA'
  AND ar.docente_id = 1
ORDER BY ir.nivel_riesgo DESC, ar.fecha_generacion DESC;
```

### C-20. Resumen de sesiones de tutoría y cumplimiento de compromisos

**Propósito:** Permite a gestión académica evaluar el seguimiento brindado por los tutores durante el ciclo (RF-21).

```sql
-- C-20: Resumen de tutorías por tutor y ciclo
SELECT
    d.codigo                        AS tutor_codigo,
    d.nombres + ' ' + d.apellidos   AS tutor,
    ca.periodo,
    COUNT(DISTINCT at2.asignacion_tutor_id)      AS estudiantes_asignados,
    COUNT(st.sesion_tutoria_id)                  AS total_sesiones,
    SUM(st.duracion_minutos)                     AS minutos_tutoria,
    COUNT(CASE WHEN st.compromisos IS NOT NULL
               THEN 1 END)                       AS sesiones_con_compromisos,
    AVG(st.duracion_minutos)                     AS duracion_promedio_min
FROM asignacion_tutor at2
    INNER JOIN docente d            ON d.docente_id          = at2.docente_id
    INNER JOIN ciclo_academico ca   ON ca.ciclo_academico_id = at2.ciclo_academico_id
    LEFT  JOIN sesion_tutoria st    ON st.asignacion_tutor_id = at2.asignacion_tutor_id
WHERE ca.periodo = '2026-2'
GROUP BY d.codigo, d.nombres, d.apellidos, ca.periodo
ORDER BY total_sesiones DESC;
```
