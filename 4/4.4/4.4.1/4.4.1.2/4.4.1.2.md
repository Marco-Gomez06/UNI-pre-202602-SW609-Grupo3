# 4.4.1.2. Consultas Relacionales: Módulo 2 (GOMEZ) — Docentes, secciones y aulas

Las siguientes cinco consultas fueron desarrolladas en **T-SQL** para Microsoft SQL Server 2022 por **Gomez**, cubriendo las necesidades del **Módulo 2: Docentes, secciones y aulas**.

---

### C-06. Detalle de la carga horaria semanal de un docente

**Propósito:** Permite a la gestión académica verificar que un docente no tiene secciones con horario superpuesto (RF-06).

```sql
-- C-06: Carga horaria de un docente en el ciclo activo
SELECT
    d.codigo                        AS docente_codigo,
    d.nombres + ' ' + d.apellidos   AS docente_nombre,
    c.codigo                        AS curso_codigo,
    s.codigo_seccion,
    CASE hs.dia_semana
        WHEN 1 THEN 'Lunes'    WHEN 2 THEN 'Martes'
        WHEN 3 THEN 'Miércoles'WHEN 4 THEN 'Jueves'
        WHEN 5 THEN 'Viernes'  WHEN 6 THEN 'Sábado'
    END                             AS dia,
    FORMAT(CAST(hs.hora_inicio AS DATETIME), 'HH:mm') AS hora_inicio,
    FORMAT(CAST(hs.hora_fin    AS DATETIME), 'HH:mm') AS hora_fin,
    hs.tipo_sesion,
    a.codigo                        AS aula
FROM docente d
    INNER JOIN seccion s            ON s.docente_id          = d.docente_id
    INNER JOIN curso c              ON c.curso_id            = s.curso_id
    INNER JOIN ciclo_academico ca   ON ca.ciclo_academico_id = s.ciclo_academico_id
    INNER JOIN horario_seccion hs   ON hs.seccion_id         = s.seccion_id
    INNER JOIN aula a               ON a.aula_id             = s.aula_id
WHERE ca.estado  = 'ACTIVO'
  AND d.codigo   = 'D-0101'
ORDER BY hs.dia_semana, hs.hora_inicio;
```

### C-07. Historial de solicitudes de ampliación de vacantes por sección

**Propósito:** Permite a la Comisión de Matrícula revisar todas las solicitudes de ampliación de una sección, con su estado y fecha de resolución (RF-08).

```sql
-- C-07: Ampliaciones de vacantes de una sección
SELECT
    c.codigo                                    AS curso_codigo,
    s.codigo_seccion,
    us.correo_institucional                     AS solicitante,
    ua.correo_institucional                     AS aprobador,
    av.cantidad_solicitada,
    ISNULL(CAST(av.cantidad_aprobada AS VARCHAR),'—') AS cantidad_aprobada,
    av.motivo,
    av.estado,
    FORMAT(av.fecha_solicitud,  'dd/MM/yyyy HH:mm') AS fecha_solicitud,
    FORMAT(av.fecha_resolucion, 'dd/MM/yyyy HH:mm') AS fecha_resolucion
FROM ampliacion_vacantes av
    INNER JOIN seccion s    ON s.seccion_id          = av.seccion_id
    INNER JOIN curso c      ON c.curso_id            = s.curso_id
    INNER JOIN usuario us   ON us.usuario_id         = av.usuario_solicita_id
    LEFT  JOIN usuario ua   ON ua.usuario_id         = av.usuario_aprueba_id
WHERE c.codigo = 'SW609'
ORDER BY av.fecha_solicitud DESC;
```

### C-08. Función: calcular créditos matriculados por estudiante en el ciclo activo

**Propósito:** Calcula el total de créditos que tiene un estudiante en el ciclo activo, para verificar que no supera el tope de su turno (RF-13, RN-02).

```sql
-- C-08: Función escalar de créditos matriculados
CREATE OR ALTER FUNCTION fn_creditos_matriculados (
    @estudiante_id      INT,
    @ciclo_academico_id INT
)
RETURNS INT
AS
BEGIN
    DECLARE @total_creditos INT;

    SELECT @total_creditos = ISNULL(SUM(c.creditos), 0)
    FROM matricula m
        INNER JOIN seccion s    ON s.seccion_id  = m.seccion_id
        INNER JOIN curso c      ON c.curso_id    = s.curso_id
    WHERE m.estudiante_id       = @estudiante_id
      AND s.ciclo_academico_id  = @ciclo_academico_id
      AND m.estado              = 'MATRICULADO';

    RETURN @total_creditos;
END;
GO

-- Ejemplo de uso
SELECT
    e.codigo,
    e.nombres + ' ' + e.apellidos      AS estudiante,
    dbo.fn_creditos_matriculados(e.estudiante_id, 2) AS creditos_matriculados
FROM estudiante e
ORDER BY creditos_matriculados DESC;
GO
```

### C-09. Reporte de ocupación de aulas por día y hora

**Propósito:** Permite a la gestión académica identificar aulas disponibles para asignar nuevas secciones (RF-07).

```sql
-- C-09: Ocupación de aulas en el ciclo activo
SELECT
    a.codigo                    AS aula,
    a.nombre                    AS nombre_aula,
    a.tipo,
    a.capacidad,
    CASE hs.dia_semana
        WHEN 1 THEN 'Lunes'    WHEN 2 THEN 'Martes'
        WHEN 3 THEN 'Miércoles'WHEN 4 THEN 'Jueves'
        WHEN 5 THEN 'Viernes'  WHEN 6 THEN 'Sábado'
    END                         AS dia,
    FORMAT(CAST(hs.hora_inicio AS DATETIME),'HH:mm') AS desde,
    FORMAT(CAST(hs.hora_fin    AS DATETIME),'HH:mm') AS hasta,
    c.codigo                    AS curso_ocupante,
    s.codigo_seccion
FROM aula a
    LEFT JOIN seccion s             ON s.aula_id             = a.aula_id
    LEFT JOIN ciclo_academico ca    ON ca.ciclo_academico_id = s.ciclo_academico_id
                                    AND ca.estado            = 'ACTIVO'
    LEFT JOIN horario_seccion hs    ON hs.seccion_id         = s.seccion_id
    LEFT JOIN curso c               ON c.curso_id            = s.curso_id
WHERE a.facultad_id = 1
ORDER BY a.codigo, hs.dia_semana, hs.hora_inicio;
```

### C-10. Procedimiento: registrar ampliación de vacantes y actualizar sección

**Propósito:** Registra una ampliación aprobada en la tabla `ampliacion_vacantes` y actualiza de forma atómica el contador de vacantes de la sección (RF-08, RN-08).

```sql
-- C-10: Procedimiento para aprobar ampliación de vacantes
CREATE OR ALTER PROCEDURE sp_aprobar_ampliacion
    @ampliacion_vacantes_id INT,
    @usuario_aprueba_id     INT,
    @cantidad_aprobada      INT
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRANSACTION;
    BEGIN TRY
        -- Actualizar la solicitud
        UPDATE ampliacion_vacantes
        SET
            cantidad_aprobada   = @cantidad_aprobada,
            usuario_aprueba_id  = @usuario_aprueba_id,
            estado              = 'APROBADA',
            fecha_resolucion    = GETDATE()
        WHERE ampliacion_vacantes_id = @ampliacion_vacantes_id
          AND estado                 = 'PENDIENTE';

        IF @@ROWCOUNT = 0
            THROW 50001, 'La solicitud no existe o ya fue resuelta.', 1;

        -- Actualizar vacantes en la sección
        UPDATE s
        SET vacantes_totales     = s.vacantes_totales     + @cantidad_aprobada,
            vacantes_disponibles = s.vacantes_disponibles + @cantidad_aprobada
        FROM seccion s
            INNER JOIN ampliacion_vacantes av
                ON av.seccion_id = s.seccion_id
        WHERE av.ampliacion_vacantes_id = @ampliacion_vacantes_id;

        -- Registrar en auditoría
        INSERT INTO auditoria (usuario_id, tabla_afectada, registro_afectado, operacion, fecha_hora, detalle)
        VALUES (@usuario_aprueba_id, 'ampliacion_vacantes',
                CAST(@ampliacion_vacantes_id AS VARCHAR),
                'UPDATE', GETDATE(),
                CONCAT('Ampliación aprobada. Vacantes añadidas: ', @cantidad_aprobada));

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        THROW;
    END CATCH;
END;
GO
```
